# MetroThemePlasma — Windows 8.1 bildirimleri (toast). AeroThemePlasma'nın bildirim uygulamacığından
# (Plasma'nın org.kde.plasma.notifications'ı temel alınarak) çatallanmıştır.
cmake_minimum_required(VERSION 3.16)
project(metrobildirim)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ECM 6.0.0 REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
set(CMAKE_AUTOMOC ON)
include(KDEInstallDirs)
include(KDECMakeSettings)
include(KDECompilerSettings NO_POLICY_SCOPE)
remove_definitions(-DQT_NO_CAST_FROM_ASCII)
include(ECMQtDeclareLoggingCategory)
include(ECMQmlModule)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Quick DBus WaylandClient)
find_package(Qt6WaylandClient COMPONENTS Private)
find_package(Wayland REQUIRED COMPONENTS Client)
find_package(LibNotificationManager REQUIRED)
find_package(KF6 REQUIRED COMPONENTS I18n CoreAddons Config GlobalAccel KIO Notifications)
find_package(Plasma REQUIRED)
find_package(PlasmaQuick REQUIRED)
find_package(PlasmaWaylandProtocols 1.23.0 REQUIRED)
add_subdirectory(applet)
