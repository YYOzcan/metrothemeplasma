# MetroThemePlasma Qt stili ("metro"): Kvantum Metro + Windows 8.1 menüleri (QProxyStyle eklentisi)
cmake_minimum_required(VERSION 3.20)
project(metrostil LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_package(KF6WindowSystem REQUIRED)
add_library(metrostil MODULE metrostil.cpp)
target_link_libraries(metrostil PRIVATE Qt6::Widgets KF6::WindowSystem)
set_target_properties(metrostil PROPERTIES PREFIX "" OUTPUT_NAME "metrostil")
